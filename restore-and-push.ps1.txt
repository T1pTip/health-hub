# Restore index.html from base64+gzip
$ErrorActionPreference = 'Stop'
Set-Location $PSScriptRoot

Write-Host "=== Step 1: Decoding b64gz to index.html ===" -ForegroundColor Cyan
$b64 = Get-Content -Raw "index.html.b64gz"
$bytes = [Convert]::FromBase64String($b64)
$ms = [System.IO.MemoryStream]::new($bytes)
$gz = [System.IO.Compression.GZipStream]::new($ms, [System.IO.Compression.CompressionMode]::Decompress)
$out = [System.IO.MemoryStream]::new()
$gz.CopyTo($out)
$html = [System.Text.Encoding]::UTF8.GetString($out.ToArray())
Set-Content -Path "index.html" -Value $html -Encoding UTF8 -NoNewline

Write-Host "✓ index.html created: $((Get-Item index.html).Length) bytes" -ForegroundColor Green
$expectedMD5 = "7cb39d24e690208043914eeef22ebb15"
$actualMD5 = (Get-FileHash index.html -Algorithm MD5).Hash.ToLower()
if ($actualMD5 -eq $expectedMD5) {
    Write-Host "✓ MD5 verified: $actualMD5" -ForegroundColor Green
} else {
    Write-Host "✗ MD5 mismatch! Expected $expectedMD5, got $actualMD5" -ForegroundColor Red
    exit 1
}

Write-Host "`n=== Step 2: Cleanup transient files ===" -ForegroundColor Cyan
Remove-Item "index.html.b64gz" -Force
Remove-Item "restore-and-push.ps1" -Force -ErrorAction SilentlyContinue  # self-delete (optional)
Write-Host "✓ Cleaned up b64gz" -ForegroundColor Green

Write-Host "`n=== Step 3: Git commit + push ===" -ForegroundColor Cyan
git add index.html
git commit -m "v2.0.0: Menu editing + workout editing + blood test persona"
git push origin master

Write-Host "`n🎉 Done! Site live at https://t1ptip.github.io/health-hub/" -ForegroundColor Green
